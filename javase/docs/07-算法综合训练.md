# 第7章：算法综合训练

## 知识点

本章为综合算法训练，核心思想：

### 1. 双指针（快慢指针）
- 慢指针指向"存入位置"，快指针负责"找数据"
- 典型应用：原地删除元素、数组去重

### 2. 随机分配（预留思想）
- 随机分配总额时，每份至少留最小值，并为后续预留最小份额，最后一份取剩余

### 3. 二路归并
- 两个有序数组用两个指针比较，小的先放入结果数组；某数组耗尽后另一个直接追加

### 4. 计数统计
- 用数组作计数器，下标代表类别，值代表次数

### 5. 动态规划思想（接雨水）
- 大问题拆小问题：分别从左往右、从右往左记录"柱子+水"高度，取交集（逐位取小）再减柱子高度

### 1. 移除元素（双指针）

源文件：`RemoveElement.java`

```java
public class RemoveElement {
    public static void main(String[] args) {
        // 移除元素：快慢指针删除所有等于val的元素
        int[] nums = {0, 1, 2, 2, 3, 0, 4, 2};
        int val = 2;

        int slow = 0;
        for (int fast = 0; fast < nums.length; fast++) {
            if (nums[fast] != val) {
                nums[slow] = nums[fast];
                slow++;
            }
        }

        System.out.println("剩余 " + slow + " 个元素：");
        for (int i = 0; i < slow; i++) {
            System.out.print(nums[i] + " ");
        }
        System.out.println();
    }
}
```

### 2. 随机红包分配（预留思想）

源文件：`RedPacket.java`

```java
import java.util.Random;
import java.util.Scanner;

public class RedPacket {
    public static void main(String[] args) {
        // 红包问题：M元N个红包，每人最少1分，并为后面的人预留1分
        Scanner sc = new Scanner(System.in);
        int totalFen;
        while (true) {
            System.out.println("请输入红包总金额（元）：");
            totalFen = sc.nextInt() * 100;
            if (totalFen > 0) {
                break;
            }
            System.out.println("金额必须大于0，请重新输入");
        }

        int count;
        while (true) {
            System.out.println("请输入红包个数：");
            count = sc.nextInt();
            if (count > 0 && totalFen >= count) {
                break;
            }
            System.out.println("红包个数不合法（每人最少1分钱），请重新输入");
        }

        Random r = new Random();
        int remain = totalFen;

        for (int i = 1; i <= count; i++) {
            int money;
            if (i == count) {
                // 最后一个人拿剩余全部
                money = remain;
            } else {
                // 每人最少1分，且给后面每人预留1分
                int max = remain - (count - i);
                money = r.nextInt(max) + 1;
            }
            remain = remain - money;
            System.out.println("第" + i + "个人抢到红包：" + (money / 100.0) + "元");
        }
    }
}
```

### 3. 两个正序数组的中位数（二路归并）

源文件：`Median.java`

```java
public class Median {
    public static void main(String[] args) {
        // 中位数：二路归并合并两个正序数组，求中位数
        int[] arr1 = {1, 3, 5, 7};
        int[] arr2 = {2, 4, 6, 8, 9};

        int[] merged = merge(arr1, arr2);

        System.out.print("合并后数组：");
        for (int i = 0; i < merged.length; i++) {
            System.out.print(merged[i] + " ");
        }
        System.out.println();

        double median;
        int mid = merged.length / 2;
        if (merged.length % 2 == 1) {
            median = merged[mid];
        } else {
            median = (merged[mid - 1] + merged[mid]) / 2.0;
        }
        System.out.println("中位数：" + median);
    }

    public static int[] merge(int[] arr1, int[] arr2) {
        int[] result = new int[arr1.length + arr2.length];
        int i = 0;
        int j = 0;
        int k = 0;
        while (i < arr1.length && j < arr2.length) {
            if (arr1[i] <= arr2[j]) {
                result[k] = arr1[i];
                i++;
            } else {
                result[k] = arr2[j];
                j++;
            }
            k++;
        }
        while (i < arr1.length) {
            result[k] = arr1[i];
            i++;
            k++;
        }
        while (j < arr2.length) {
            result[k] = arr2[j];
            j++;
            k++;
        }
        return result;
    }
}
```

### 4. 学生会投票统计（计数+随机模拟）

源文件：`VoteStats.java`

```java
import java.util.Random;

public class VoteStats {
    public static void main(String[] args) {
        // 统计投票：1000人投票，0弃权，1-5候选人，统计得票数和得票率
        Random r = new Random();
        // votes[0]弃权票，votes[1]~votes[5]候选人得票
        int[] votes = new int[6];
        int total = 1000;

        for (int i = 0; i < total; i++) {
            int vote = r.nextInt(6);
            votes[vote]++;
        }

        int maxIndex = 1;
        for (int i = 1; i <= 5; i++) {
            System.out.println(i + "号候选人得票数：" + votes[i]
                    + "，得票率：" + (votes[i] * 100.0 / total) + "%");
            if (votes[i] > votes[maxIndex]) {
                maxIndex = i;
            }
        }
        System.out.println("得票最多的候选人：" + maxIndex + "号");

        System.out.println("弃票数：" + votes[0]
                + "，弃票率：" + (votes[0] * 100.0 / total) + "%");
    }
}
```

### 5. 接雨水（动态规划思想）

源文件：`TrappingRain.java`

```java
public class TrappingRain {
    public static void main(String[] args) {
        // 接雨水：左右扫描取交集（取min），减去柱子高度
        int[] height = {0, 1, 0, 2, 1, 0, 1, 3, 2, 1, 2, 1};

        // 从左往右：记录每个位置左边（含自己）最高的柱子
        int[] left = new int[height.length];
        int max = 0;
        for (int i = 0; i < height.length; i++) {
            if (height[i] > max) {
                max = height[i];
            }
            left[i] = max;
        }

        // 从右往左：记录每个位置右边（含自己）最高的柱子
        int[] right = new int[height.length];
        max = 0;
        for (int i = height.length - 1; i >= 0; i--) {
            if (height[i] > max) {
                max = height[i];
            }
            right[i] = max;
        }

        // 每个位置的雨水 = min(左最高, 右最高) - 柱子高度
        int water = 0;
        for (int i = 0; i < height.length; i++) {
            int level = Math.min(left[i], right[i]);
            water = water + (level - height[i]);
        }

        System.out.println("能接的雨水：" + water + " 个单位");
    }
}
```

### 6. 大乐透选号与中奖判定

源文件：`Lottery.java`

```java
import java.util.Random;

public class Lottery {
    public static void main(String[] args) {
        // 大乐透：前区01-35选5个不重复，后区01-12选2个不重复，跨区可重复
        int[] myFront = generateNumbers(5, 35);
        int[] myBack = generateNumbers(2, 12);
        System.out.print("我的投注  前区：");
        printNumbers(myFront);
        System.out.print("          后区：");
        printNumbers(myBack);

        int[] winFront = generateNumbers(5, 35);
        int[] winBack = generateNumbers(2, 12);
        System.out.print("开奖号码  前区：");
        printNumbers(winFront);
        System.out.print("          后区：");
        printNumbers(winBack);

        int frontHit = countHit(myFront, winFront);
        int backHit = countHit(myBack, winBack);
        System.out.println("命中：前区" + frontHit + "个，后区" + backHit + "个");
        System.out.println("中奖结果：" + checkPrize(frontHit, backHit));
    }

    // 从1~max中随机抽取count个不重复号码
    public static int[] generateNumbers(int count, int max) {
        Random r = new Random();
        int[] numbers = new int[count];
        int size = 0;
        while (size < count) {
            int number = r.nextInt(max) + 1;
            boolean exists = false;
            for (int i = 0; i < size; i++) {
                if (numbers[i] == number) {
                    exists = true;
                    break;
                }
            }
            if (!exists) {
                numbers[size] = number;
                size++;
            }
        }
        return numbers;
    }

    // 统计两组号码命中的个数
    public static int countHit(int[] mine, int[] win) {
        int hit = 0;
        for (int i = 0; i < mine.length; i++) {
            for (int j = 0; j < win.length; j++) {
                if (mine[i] == win[j]) {
                    hit++;
                    break;
                }
            }
        }
        return hit;
    }

    // 根据中奖规则判断奖级
    public static String checkPrize(int front, int back) {
        if (front == 5 && back == 2) {
            return "一等奖";
        } else if (front == 5 && back == 1) {
            return "二等奖";
        } else if ((front == 5 && back == 0) || (front == 4 && back == 2)) {
            return "三等奖";
        } else if ((front == 4 && back == 1) || (front == 3 && back == 2)) {
            return "四等奖";
        } else if ((front == 4 && back == 0) || (front == 3 && back == 1) || (front == 2 && back == 2)) {
            return "五等奖";
        } else if ((front == 3 && back == 0) || (front == 1 && back == 2)
                || (front == 2 && back == 1) || (front == 0 && back == 2)) {
            return "六等奖";
        } else {
            return "未中奖";
        }
    }

    // 打印号码，不足两位补0
    public static void printNumbers(int[] numbers) {
        for (int i = 0; i < numbers.length; i++) {
            if (numbers[i] < 10) {
                System.out.print("0" + numbers[i] + " ");
            } else {
                System.out.print(numbers[i] + " ");
            }
        }
        System.out.println();
    }
}
```
